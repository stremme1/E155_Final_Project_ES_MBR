# Complete Pin Assignment Constraints File
# Copy this to your Radiant project as constraints.pcf
#
# NOTE: Clock is generated internally by HSOSC - DO NOT assign clk to a pin!

# ============================================================================
# Top-Level Module Port Order Reference
# ============================================================================
# This table shows all ports in the order they appear in drum_set_top.sv
#
# INPUTS (in order):
# | Signal Name      | Pin  | Description                          | Pull-up |
# |------------------|------|--------------------------------------|---------|
# | rst_n            | P43  | Reset button (active LOW)            | 10kΩ    |
# | miso1            | P12  | SPI MISO from Sensor 1 (Right Hand)  | No      |
# | miso2            | P6   | SPI MISO from Sensor 2 (Left Hand)  | No      |
# | calib_button     | P11  | Calibration button                   | 10kΩ    |
# | kick_button      | P2   | Kick drum button (optional)          | 10kΩ    |
#
# OUTPUTS (in order):
# | Signal Name      | Pin  | Description                          | Pull-up |
# |------------------|------|--------------------------------------|---------|
# | sclk1            | P20  | SPI clock to Sensor 1                | No      |
# | mosi1            | P13  | SPI MOSI to Sensor 1                 | No      |
# | cs_n1            | P18  | Chip select to Sensor 1 (active LOW) | 10kΩ    |
# | sclk2            | P4   | SPI clock to Sensor 2                | No      |
# | mosi2            | P47  | SPI MOSI to Sensor 2                 | No      |
# | cs_n2            | P48  | Chip select to Sensor 2 (active LOW) | 10kΩ    |
# | mcu_sclk         | P21  | SPI clock to MCU                     | No      |
# | mcu_mosi         | P10  | SPI MOSI to MCU                      | No      |
# | mcu_cs_n         | P19  | Chip select to MCU (active LOW)      | 10kΩ    |
# | led_initialized  | P28  | LED when initialized                 | No      |
# | led_error        | P38  | LED for errors                       | No      |
#
# ============================================================================
# Pin Assignment Constraints (for copy-paste into Radiant)
# ============================================================================

# System Reset
# Note: Clock is generated internally by HSOSC - no clock pin needed
set_io rst_n   P43           # Reset button (active LOW - goes LOW to reset)

# BNO085 Sensor 1 (Right Hand)
set_io sclk1   P20           # SPI clock
set_io mosi1   P13           # SPI MOSI
set_io miso1   P12           # SPI MISO
set_io cs_n1   P18           # Chip select (needs 10kΩ pull-up)

# BNO085 Sensor 2 (Left Hand)
set_io sclk2   P4            # SPI clock
set_io mosi2   P47           # SPI MOSI
set_io miso2   P6            # SPI MISO
set_io cs_n2   P48           # Chip select (needs 10kΩ pull-up)

# User Interface Buttons
set_io calib_button  P11     # Calibration button (left button, needs 10kΩ pull-up)
set_io kick_button    P2     # Kick button (right button, needs 10kΩ pull-up)

# MCU SPI Communication
set_io mcu_sclk      P21     # SPI clock to MCU
set_io mcu_mosi      P10     # SPI MOSI to MCU
set_io mcu_cs_n      P19     # Chip select to MCU (needs 10kΩ pull-up)

# Status LEDs
set_io led_initialized P28   # LED when initialized (HIGH when both sensors ready)
set_io led_error       P38   # LED for errors (HIGH when sensor error detected)
